function ClockWidget() {
  const [now, setNow] = useState(new Date());
  useEffect(() => {
    const t = setInterval(() => setNow(new Date()), 1000);
    return () => clearInterval(t);
  }, []);
  return (
    <div style={{ fontSize: "14px", lineHeight: 1.3 }}>
      <div style={{ fontWeight: 700 }}>{now.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}</div>
      <div style={{ color: "var(--t3)", fontSize: "12px" }}>{now.toLocaleDateString(undefined, { weekday: "short", month: "short", day: "numeric" })}</div>
      <div style={{ color: "var(--t3)", fontSize: "12px" }}>Atlanta</div>
    </div>
  );
}

function RailColumn({ children, gridColumn = "span 3" }) {
  return (
    <div style={{ gridColumn, display: "flex", flexDirection: "column", gap: 14, minWidth: 0 }}>
      {children}
    </div>
  );
}

const GridCard = ({ title, icon, actions, children, style: styleProp, gridColumn }) => {
  const [isHovered, setIsHovered] = useState(false);

  const hoverGlow = isHovered 
    ? "0 0 0 2px rgba(255, 0, 13, 0.32), 0 10px 30px rgba(255, 0, 13, 0.2)" 
    : "none";

  return (
    <div
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
      style={{
        gridColumn: gridColumn === false ? undefined : (gridColumn || "span 4"),
        position: "relative",
        borderRadius: "16px",
        padding: "3px",
        border: isHovered ? "1px solid rgba(255, 0, 13, 0.3)" : "1px solid rgba(255,255,255,0.09)",
        transition: "all 0.2s ease",
        boxShadow: hoverGlow,
        ...styleProp,
      }}
    >
      <div className="liquid-glass" style={{
        padding: "16px 18px",
        minHeight: "auto",
        display: "flex",
        flexDirection: "column",
        borderRadius: "14px",
      }}>
        {(title || icon || actions) && (
          <div style={{
            display: "flex",
            alignItems: "center",
            gap: 10,
            paddingBottom: 10,
            borderBottom: "1px solid rgba(255,255,255,0.08)",
            marginBottom: 10,
            flexShrink: 0,
          }}>
            {icon && <span style={{ fontSize: 18, color: "#ff000d" }}>{icon}</span>}
            {title && (
              <span style={{
                flex: 1,
                fontSize: "14.5px",
                fontWeight: 700,
                letterSpacing: ".04em",
                textTransform: "uppercase",
                color: "#ff000d",
              }}>
                {title}
              </span>
            )}
            {actions}
          </div>
        )}
        <div style={{ flex: 1, overflow: "hidden", fontSize: "14px", lineHeight: "1.45" }}>
          {children}
        </div>
      </div>
    </div>
  );
};

          </div>

        <GridCard title="Product Revenue" icon={<LucideLineChart size={18} />} gridColumn="span 5" actions={
          <Button variant={editProducts ? "primary" : "ghost"} style={{ padding: "6px 12px", fontSize: "13px" }} onClick={saveProducts}>
            {editProducts ? "SAVE" : "EDIT"}
          </Button>
        }>
          <div style={{ fontSize: "13px", color: "var(--t3)", marginBottom: 6 }}>Sources: KPI Revenue</div>
          <div style={{ fontSize: "14px", display: "grid", gridTemplateColumns: "1.2fr 0.8fr 0.8fr", gap: 8 }}>
            {products.map((p, i) => (
              <React.Fragment key={i}>
                <div style={{ fontWeight: 600 }}>
                  {editProducts ? <input value={productInputs[`${i}-name`] ?? p.name} onChange={e => setProductInputs(pi => ({...pi, [`${i}-name`]: e.target.value}))} style={{width:"100%", fontSize:"14px"}} /> : p.name}

                  {editProducts ? <input type="number" value={productInputs[`${i}-moRev`] ?? p.moRev} onChange={e => setProductInputs(pi => ({...pi, [`${i}-moRev`]: e.target.value}))} style={{width:"100%", fontSize:"14px"}} /> : "$" + p.moRev}
                </div>
                <div style={{ textAlign: "right", color: "var(--t3)" }}>
                  {editProducts ? <input type="number" value={productInputs[`${i}-ytdRev`] ?? p.ytdRev} onChange={e => setProductInputs(pi => ({...pi, [`${i}-ytdRev`]: e.target.value}))} style={{width:"100%", fontSize:"14px"}} /> : "$" + p.ytdRev}

        </GridCard>

        <GridCard title="Financial Balances" icon={<Wallet size={18} />} gridColumn="span 4" actions={
          <Button variant={editFinance ? "primary" : "ghost"} style={{ padding: "6px 12px", fontSize: "13px" }} onClick={saveFinanceInstitutions}>
            {editFinance ? "SAVE" : "EDIT"}
          </Button>
        }>
          <div style={{ fontSize: "14px" }}>
            {financeInstitutions.map((inst, i) => (
              <div key={i} style={{ display: "flex", justifyContent: "space-between", marginBottom: 4 }}>

                <span style={{ color: "var(--crimson)" }}>
                  ${(editFinance ? Number(financeInputs[`${i}-checking`] ?? inst.checking) : inst.checking) + (editFinance ? Number(financeInputs[`${i}-savings`] ?? inst.savings) : inst.savings)}
                </span>

          </div>

        <GridCard title="Credit Score" icon={<LucideBarChart3 size={18} />} gridColumn="span 3" actions={

            {editCredit ? "SAVE" : "EDIT"}

        }>
          <div style={{ display: "flex", alignItems: "center", gap: 12, fontSize: "14px" }}>
            <div style={{ position: "relative", width: "100%", height: 32 }}>
              <div style={{ position: "absolute", bottom: 0, left: 0, width: "100%", height: 8, background: "linear-gradient(to right, #ff4f5e, #888, #ff000d)", borderRadius: 4 }} />

            </div>
          </div>
          <div style={{ fontSize: "13px", color: "var(--t3)", marginTop: 8 }}>Karma: {karmaScore || "—"} · Experian: {experScore || "—"}</div>
        </GridCard>

        <GridCard title="Budget (Finance Hub)" icon={<Wallet size={18} />} gridColumn="span 4">
          <div style={{ fontSize: "14px", color: "var(--t2)" }}>
            {(financeSnap.budget?.categories || []).slice(0, 5).map((c, i) => (
              <div key={i} style={{ display: "flex", justifyContent: "space-between", marginBottom: 4 }}>

              </div>

            {!(financeSnap.budget?.categories || []).length && <div>Set categories in Finance Hub</div>}
          </div>
        </GridCard>

        <GridCard title="Low-effort Money Tips" icon={<Bot size={18} />} gridColumn="span 6">
          <div style={{ fontSize: "14px", color: "var(--t2)", display: "flex", flexDirection: "column", gap: 6 }}>
            {getDailyMoneyTips().map((tip, i) => (
              <div key={i} style={{ display: "flex", justifyContent: "space-between" }}>

              </div>
            ))}
          </div>
        </GridCard>

        <GridCard title="Life Hacks" icon={<BookOpen size={18} />} gridColumn="span 6">
          <div style={{ fontSize: "14px", color: "var(--t2)", display: "flex", flexDirection: "column", gap: 6 }}>
            {getDailyLifeHacks().map((h, i) => (
              <div key={i} style={{ display: "flex", justifyContent: "space-between" }}>
                <span style={{ flex: 1 }}>{h}</span>
                <button type="button" style={{ fontSize: "13px", border: "1px solid var(--crimson)", color: "var(--crimson)", background: "transparent", padding: "2px 8px", cursor: "pointer", borderRadius: 4 }}>+</button>
              </div>
            ))}
          </div>

        <GridCard title="Social Hub" icon={<Share2 size={18} />} gridColumn="span 3">
          <div style={{ fontSize: "14px", color: "var(--t2)" }}>
            {social ? [
              social.meta?.instagram ? `IG ${social.meta.instagram.followers ?? "—"}` : null,
              social.x ? `X ${social.x.followers ?? "—"}` : null,
              social.yt ? `YT ${social.yt.subscribers ?? "—"}` : null,
            ].filter(Boolean).map((s, i) => <div key={i} style={{ marginBottom: 6 }}>{s}</div>) : "Loading…"}
          </div>
        </GridCard>

        <GridCard title="Playlist" icon={<Music size={18} />} gridColumn="span 3">
          <MusicPlaylistMini />
        </GridCard>

        <GridCard title="Quick budget" icon={<Wallet size={18} />} gridColumn="span 6">
          <div style={{ display: "flex", gap: 10, marginBottom: 12 }}>
            <Input placeholder="Desc" value={budgetDesc} onChange={e => setBudgetDesc(e.target.value)} style={{ fontSize: "14px", padding: "8px 12px", flex: 2 }} />
            <Input placeholder="$" type="number" value={budgetAmount} onChange={e => setBudgetAmount(e.target.value)} style={{ fontSize: "14px", padding: "8px 12px", flex: 1 }} />
            <Button style={{ padding: "8px 16px", fontSize: "14px" }} onClick={() => addBudget(budgetDesc, budgetAmount)}>+</Button>

          <div style={{ fontSize: "14px", maxHeight: 140, overflow: "auto" }}>
            {budgetItems.map(item => (
