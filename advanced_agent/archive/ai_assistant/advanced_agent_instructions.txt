import json
from typing import Dict, List, Any
from datetime import datetime
from config import Config

class AgentMemory:
    def __init__(self):
        self.memory_file = Config.MEMORY_FILE
        self.short_term: List[Dict] = []   # last 10 interactions
        self.long_term: Dict[str, Any] = self._load()
    
    def _load(self) -> Dict:
        try:
            with open(self.memory_file, "r") as f:
                return json.load(f)
        except FileNotFoundError:
            return {"conversations": [], "facts": [], "preferences": {}}
    
    def save(self):
        with open(self.memory_file, "w") as f:
            json.dump(self.long_term, f, indent=2)
    
    def add_interaction(self, user_input: str, response: str, model_used: str):
        entry = {
            "timestamp": datetime.now().isoformat(),
            "user": user_input,
            "assistant": response,
            "model": model_used
        }
        self.short_term.append(entry)
        if len(self.short_term) > 10:
            self.long_term["conversations"].extend(self.short_term[:-10])
            self.short_term = self.short_term[-10:]
        self.save()
    
    def get_context(self, limit: int = 5) -> str:
        recent = self.short_term[-limit:] if self.short_term else []
        context = "\n".join([f"User: {c['user']}\nAI: {c['assistant']}" for c in recent])
        return context

11. agent/router.py
python
from typing import Dict, Any
from models.claude import ClaudeModel
from models.gpt import GPTModel
from models.gemini import GeminiModel
from models.grok import GrokModel
from models.perplexity import PerplexityModel
from models.suno import SunoModel

class ModelRouter:
    def __init__(self):
        self.models = {
            "claude": ClaudeModel(),
            "gpt": GPTModel(),
            "gemini": GeminiModel(),
            "grok": GrokModel(),
            "perplexity": PerplexityModel(),
            "suno": SunoModel()
        }
    
    def select_model(self, task: str) -> str:
        # Rule-based routing for optimal performance
        task_lower = task.lower()
        if any(kw in task_lower for kw in ["search", "latest", "news", "today"]):
            return "perplexity"
        if any(kw in task_lower for kw in ["code", "program", "function", "debug"]):
            return "gpt"
        if any(kw in task_lower for kw in ["analyze", "reason", "explain"]):
            return "claude"
        if any(kw in task_lower for kw in ["song", "audio", "music", "melody"]):
            return "suno"
        if any(kw in task_lower for kw in ["image", "vision", "multimodal"]):
            return "gemini"
        if any(kw in task_lower for kw in ["humor", "joke", "funny", "roast"]):
            return "grok"
        return "claude"   # default
    
    async def route(self, task: str, context: str = "") -> Dict[str, Any]:
        model_name = self.select_model(task)
        model = self.models[model_name]
        
        if model_name == "perplexity":
            response = await model.search_and_generate(task)
        elif model_name == "suno":
            response = await model.generate_song(task)
        else:
            response = await model.generate(task, context)
        
        return {"model_used": model_name, "response": response}

python
import asyncio
from agent.router import ModelRouter
from agent.memory import AgentMemory
from agent.tools.clickup_integration import ClickUpTool
from agent.tools.make_integration import trigger_make_scenario
from agent.tools.vscode_integration import VSCodeTool
from agent.tools.cowork_integration import CoWorkTool   # stub
from utils.code_executor import execute_python

class UnifiedAgent:
    def __init__(self):
        self.router = ModelRouter()
        self.memory = AgentMemory()
        self.clickup = ClickUpTool()
        self.vscode = VSCodeTool()
        # self.cowork = CoWorkTool()
    
    async def process(self, user_input: str) -> dict:
        # 1. Retrieve memory context
        context = self.memory.get_context()
        
        # 2. Route to best model
        result = await self.router.route(user_input, context)
        
        # 3. Save interaction
        self.memory.add_interaction(user_input, result["response"], result["model_used"])
        
        # 4. Tool execution if needed (detect intent)
        if "create a task in clickup" in user_input.lower():
            # Parse task name, description
            task_name = "AI generated task"
            task_desc = result["response"]
            await self.clickup.create_task(task_name, task_desc, Config.CLICKUP_TEAM_ID)
            result["tools_used"] = ["clickup"]
        
        if "code" in user_input.lower() and "```" in result["response"]:
            # Extract code block and execute or send to VS Code
            code = self._extract_code(result["response"])
            if "run" in user_input.lower():
                output = execute_python(code)
                result["code_output"] = output
            await self.vscode.open_file("temp_agent_code.py")
            await self.vscode.apply_edit("temp_agent_code.py", code)
            result["tools_used"] = result.get("tools_used", []) + ["vscode"]
        
        return result
    
    def _extract_code(self, text: str) -> str:
        import re
        match = re.search(r"```python\n(.*?)\n```", text, re.DOTALL)
        return match.group(1) if match else ""

import subprocess
import tempfile

def execute_python(code: str) -> str:
    with tempfile.NamedTemporaryFile(mode="w", suffix=".py", delete=False) as f:
        f.write(code)
        f.flush()
        result = subprocess.run(["python", f.name], capture_output=True, text=True)
        return result.stdout + result.stderr

4. Plan‑and‑Execute Loop with instructor
agent/planner.py
python
import instructor
from openai import AsyncOpenAI
from pydantic import BaseModel, Field
from typing import List, Dict, Any, Literal
from config import Config
import asyncio

# Set up instructor with GPT-4 (or any model that supports tool calls)
client = instructor.from_openai(AsyncOpenAI(api_key=Config.OPENAI_API_KEY))

class ToolCall(BaseModel):
    tool_name: Literal["clickup_create_task", "vscode_open_file", "vscode_apply_edit", 
                       "make_trigger", "web_search", "generate_image", "execute_code"]
    arguments: Dict[str, Any] = Field(..., description="Arguments for the tool")

class Plan(BaseModel):
    reasoning: str = Field(..., description="Step‑by‑step reasoning")
    steps: List[ToolCall] = Field(..., description="Ordered list of tool calls to achieve the goal")

class PlanAndExecute:
    def __init__(self, agent_core):  # agent_core provides tool implementations
        self.agent_core = agent_core
        self.max_iterations = 5
    
    async def plan(self, user_goal: str, context: str) -> Plan:
        prompt = f"""
        You are a planning agent. Given the user goal and context, produce a structured plan.
        Available tools: clickup_create_task, vscode_open_file, vscode_apply_edit, make_trigger, web_search, generate_image, execute_code.
        User goal: {user_goal}
        Context: {context}
        """
        response = await client.chat.completions.create(
            model="gpt-4-turbo-preview",
            response_model=Plan,
            messages=[{"role": "user", "content": prompt}]
        )
        return response
    
    async def execute_step(self, tool_call: ToolCall) -> Any:
        # Map tool names to actual methods in agent_core
        tool_map = {
            "clickup_create_task": self.agent_core.clickup.create_task,
            "vscode_open_file": self.agent_core.vscode.open_file,
            "vscode_apply_edit": self.agent_core.vscode.apply_edit,
            "make_trigger": self.agent_core.trigger_make,
            "web_search": self.agent_core.web_search,
            "generate_image": self.agent_core.generate_image,
            "execute_code": self.agent_core.execute_code
        }
        func = tool_map.get(tool_call.tool_name)
        if not func:
            return f"Tool {tool_call.tool_name} not found"
        if asyncio.iscoroutinefunction(func):
            return await func(**tool_call.arguments)
        else:
            return func(**tool_call.arguments)
    
    async def run(self, user_goal: str, context: str) -> str:
        plan = await self.plan(user_goal, context)
        results = []
        for step in plan.steps:
            result = await self.execute_step(step)
            results.append(f"Executed {step.tool_name}: {result}")
            # Optional: short pause to avoid rate limits
            await asyncio.sleep(0.5)
        final_report = f"Plan reasoning: {plan.reasoning}\n" + "\n".join(results)
        return final_report
5. Workshop Training – Fine‑tune Router on User Feedback
agent/workshop_trainer.py
python
import json
from typing import Dict, List
from collections import defaultdict

class WorkshopTrainer:
    def __init__(self, router):
        self.router = router
        self.feedback_log: List[Dict] = []   # each: {"task": str, "model_used": str, "rating": 1-5, "better_model": str}
        self.routing_weights = defaultdict(lambda: defaultdict(float))
        self.load_weights()
    
    def load_weights(self):
        try:
            with open("routing_weights.json", "r") as f:
                self.routing_weights = json.load(f)
        except FileNotFoundError:
            pass
    
    def save_weights(self):
        with open("routing_weights.json", "w") as f:
            json.dump(self.routing_weights, f, indent=2)
    
    def collect_feedback(self, task: str, model_used: str, rating: int, better_model: str = None):
        self.feedback_log.append({
            "task": task,
            "model_used": model_used,
            "rating": rating,
            "better_model": better_model
        })
        # Update weights (simple heuristic)
        task_keywords = self._extract_keywords(task)
        for kw in task_keywords:
            # Increase weight for the successful model
            self.routing_weights[kw][model_used] = self.routing_weights[kw].get(model_used, 0) + rating / 5.0
            if better_model and better_model != model_used:
                self.routing_weights[kw][better_model] = self.routing_weights[kw].get(better_model, 0) + 1.0
        self.save_weights()
    
    def _extract_keywords(self, task: str) -> List[str]:
        # Very basic – you can use TF‑IDF or LLM for production
        words = task.lower().split()
        # Keep only potentially meaningful words
        stopwords = {"a", "an", "the", "and", "or", "but", "for", "of", "to", "in", "on", "at"}
        keywords = [w for w in words if w not in stopwords and len(w) > 2][:5]
        return keywords
    
    def update_router_logic(self):
        # Modify router's select_model method dynamically using accumulated weights
        original_select = self.router.select_model
        
        def weighted_select(task: str) -> str:
            keywords = self._extract_keywords(task)
            scores = defaultdict(float)
            for kw in keywords:
                if kw in self.routing_weights:
                    for model, weight in self.routing_weights[kw].items():
                        scores[model] += weight
            if scores:
                best_model = max(scores, key=scores.get)
                return best_model
            else:
                return original_select(task)
        
        self.router.select_model = weighted_select

Multi‑modal: The agent can now accept image bytes or audio bytes via new endpoints (add /analyze_image and /transcribe_audio) that call GeminiModel.

Long‑term memory: Every conversation older than the last 10 turns is embedded and stored in Chroma. When a query is similar to past topics, the vector store returns relevant context.

Workshop training: After each interaction, the user can rate the model used. The trainer adjusts routing keywords so that the router improves over time.
